  // Chassis
  const bodyGeo = new THREE.BoxGeometry(2.4, 0.9, 1.3);

  const headMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
  const headL = new THREE.Mesh(lightGeo, headMat);

  scene.add(car);
  vehicles.push({
    mesh: car,
    progress: (i / 5),
    speed: 0.00075 + (i * 0.00015)
  });
}

function updateVehicles() {
  const totalLength = 72 + 84 + 72 + 84; // Perimeter loop length
  vehicles.forEach(v => {
    v.progress = (v.progress + v.speed) % 1.0;
    const currentDist = v.progress * totalLength;

    if (currentDist < 72) {
      // Moving South: -36 to +36 along z=46
      const t = currentDist / 72;
      v.mesh.position.set(-36 + (72 * t), 0, 46);
      v.mesh.rotation.y = 0;
    } else if (currentDist < 156) {
      // Moving North along East road: z=46 to z=-38
      const t = (currentDist - 72) / 84;
      v.mesh.position.set(36, 0, 46 - (84 * t));
      v.mesh.rotation.y = Math.PI / 2;
    } else if (currentDist < 228) {
      // Moving West along North road: x=36 to x=-36 at z=-38
      const t = (currentDist - 156) / 72;
      v.mesh.position.set(36 - (72 * t), 0, -38);
      v.mesh.rotation.y = Math.PI;
    } else {
      // Moving South along West road: z=-38 to z=46
      const t = (currentDist - 228) / 84;
      v.mesh.position.set(-36, 0, -38 + (84 * t));
      v.mesh.rotation.y = -Math.PI / 2;
    }
  });
}

// Raycasting for Flat Selection
const raycaster = new THREE.Raycaster();
const mouse = new THREE.Vector2();

container.addEventListener('click', (event) => {
  const rect = renderer.domElement.getBoundingClientRect();
  mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
  mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;

  raycaster.setFromCamera(mouse, camera);
  const intersects = raycaster.intersectObjects(interactiveWindows);

  if (intersects.length > 0) {
    const target = intersects[0].object;
    if (target.userData && target.userData.flat) {
      // Reset all windows
      interactiveWindows.forEach(w => {
        w.material = w.userData.isDue ? windowMatDue : windowMatClear;
      });
      // Highlight clicked
      target.material = windowMatSelected;

      // Dispatch custom window event to sync with UI
      const selectEvent = new CustomEvent('flat-3d-selected', {
        detail: {
          flat: target.userData.flat,
          wing: target.userData.wing,
          floor: target.userData.floor
        }
      });
      window.dispatchEvent(selectEvent);
    }
  }
});

// Interactive Orbit / Drag rotation
let isDragging = false;
let previousMousePosition = { x: 0, y: 0 };
let targetRotationY = 0;
let currentRotationY = 0;

container.addEventListener('mousedown', (e) => {
  isDragging = true;
  previousMousePosition = { x: e.clientX, y: e.clientY };
});

window.addEventListener('mouseup', () => {
  isDragging = false;
});

window.addEventListener('mousemove', (e) => {
  if (isDragging) {
    const deltaX = e.clientX - previousMousePosition.x;
    targetRotationY += deltaX * 0.006;
    previousMousePosition = { x: e.clientX, y: e.clientY };
  }
});

// Window Resize Handling
window.addEventListener('resize', () => {
  const w = container.clientWidth || window.innerWidth;
  const h = container.clientHeight || 550;
  camera.aspect = w / h;
  camera.updateProjectionMatrix();
  renderer.setSize(w, h);
});

// Main Animation Loop
function animate() {
  requestAnimationFrame(animate);

  // Smooth rotation dampening
  currentRotationY += (targetRotationY - currentRotationY) * 0.05;
  scene.rotation.y = currentRotationY;

  // Orbit camera slight gentle sway when idle
  if (!isDragging) {
    targetRotationY += 0.0006;
  }

  updateVehicles();
  renderer.render(scene, camera);
}

animate();

})();

function updateVehicles() {
  const totalLength = 72 + 84 + 72 + 84; // Perimeter loop length
  vehicles.forEach(v => {
    v.progress = (v.progress + v.speed) % 1.0;
    const currentDist = v.progress * totalLength;

    if (currentDist < 72) {
      // Moving South: -36 to +36 along z=46
      const t = currentDist / 72;
      v.mesh.position.set(-36 + (72 * t), 0, 46);
      v.mesh.rotation.y = 0;
    } else if (currentDist < 156) {
      // Moving North along East road: z=46 to z=-38
      const t = (currentDist - 72) / 84;
      v.mesh.position.set(36, 0, 46 - (84 * t));
      v.mesh.rotation.y = Math.PI / 2;
    } else if (currentDist < 228) {
      // Moving West along North road: x=36 to x=-36 at z=-38
      const t = (currentDist - 156) / 72;
      v.mesh.position.set(36 - (72 * t), 0, -38);
      v.mesh.rotation.y = Math.PI;
    } else {
      // Moving South along West road: z=-38 to z=46
      const t = (currentDist - 228) / 84;
      v.mesh.position.set(-36, 0, -38 + (84 * t));
      v.mesh.rotation.y = -Math.PI / 2;
    }
  });
}

  const intersects = raycaster.intersectObjects(interactiveWindows);

  if (intersects.length > 0) {
    const target = intersects[0].object;
    if (target.userData && target.userData.flat) {
      // Reset all windows
      interactiveWindows.forEach(w => {
        w.material = w.userData.isDue ? windowMatDue : windowMatClear;

      // Highlight clicked

        detail: {
          flat: target.userData.flat,
          wing: target.userData.wing,
          floor: target.userData.floor
        }
      });
      window.dispatchEvent(selectEvent);
    }
  }

        const cardCategories = card.getAttribute('data-category') || '';
        if (filterValue === 'all' || cardCategories.includes(filterValue)) {
          card.style.display = 'flex';
        } else {
          card.style.display = 'none';
        }
      });
    });
  });

  // Client-side Notice Search

    const term = e.target.value.toLowerCase().trim();

      if (text.includes(term)) {
        card.style.display = 'flex';
      } else {
        card.style.display = 'none';

      const inspectorStatusBadge = document.getElementById('inspectorStatusBadge');

        });
      });

      // Action Handlers on Table Rows (Verify & Credit / Reject)
      const verifyBtns = document.querySelectorAll('.btn-verify');
      const rejectBtns = document.querySelectorAll('.btn-reject');

      verifyBtns.forEach(btn => {
        btn.addEventListener('click', (e) => {
          const row = e.target.closest('tr');
          const flat = row.getAttribute('data-flat');
          const actionContainer = row.querySelector('.action-container');
          
          btn.innerHTML = '<span class="material-symbols-outlined animate-spin text-[14px]">progress_activity</span> Crediting...';
          setTimeout(() => {
            row.classList.remove('bg-gold-soft/20');
            row.classList.add('bg-surface-soft/40');
            if (actionContainer) {
              actionContainer.innerHTML = '<span class="font-code-md text-[12px] text-success font-semibold flex items-center gap-1"><span class="material-symbols-outlined text-[16px]">check_circle</span> Credited #RP-' + Math.floor(1000 + Math.random() * 9000) + '</span>';
            }
          }, 800);
        });
      });

      rejectBtns.forEach(btn => {
        btn.addEventListener('click', (e) => {
          const row = e.target.closest('tr');
          const actionContainer = row.querySelector('.action-container');
          if (confirm('Are you sure you want to flag this UTR submission for resident re-upload?')) {
            row.style.opacity = '0.5';
            if (actionContainer) {
              actionContainer.innerHTML = '<span class="font-label-sm text-[12px] text-error font-semibold">Rejected &amp; SMS Sent</span>';
            }
          }
        });
      });

      // Inspector Direct Verification

          inspectVerifyBtn.classList.add('bg-success');
        }, 900);
      });

      // Filter search
      const searchInput = document.getElementById('searchInput');
      searchInput.addEventListener('input', (e) => {
        const query = e.target.value.toLowerCase();
        const rows = document.querySelectorAll('#queueTableBody tr');
        rows.forEach(r => {
          const text = r.innerText.toLowerCase();
          r.style.display = text.includes(query) ? '' : 'none';
        });
      });

      // Auto Match All
      document.getElementById('autoMatchAllBtn').addEventListener('click', () => {
        const btn = document.getElementById('autoMatchAllBtn');
        btn.innerHTML = '<span class="material-symbols-outlined animate-spin text-[18px]">autorenew</span> Reconciling 3 Items...';
